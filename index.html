async function startSimpleMining() { const username = document.getElementById('user').value.toLowerCase().trim(); const statusDiv = document.getElementById('mining-status'); const output = document.getElementById('output'); if (!username) { alert("Please enter a username to mine coins!"); return; } statusDiv.style.color = "#00875a"; // Reset to green statusDiv.innerText = "Rolling a 1-1000 dice..."; const luckyGuess = Math.floor(Math.random() * 1000) + 1; statusDiv.innerText = `You rolled: ${luckyGuess}! Submitting to server...`; // Call backend manually here instead of doAction to handle the 429 status code smoothly output.innerText = "Processing..."; try { const response = await fetch(WORKER_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'mine_guess', username: username, password: document.getElementById('pass').value, guess: luckyGuess.toString() }) }); const result = await response.text(); output.innerText = result; // If rate limited, turn the mining text red if (response.status === 429) { statusDiv.style.color = "#de350b"; // Red error text statusDiv.innerText = "Throttled: Too many global requests!"; } } catch (err) { output.innerText = "Connection Error: " + err.message; } }
